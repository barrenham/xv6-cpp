# Production backends have separate C++ module and archive boundaries. The
# native contracts compile these same module implementations.
function(xv_userfs_module target interface)
  add_library(${target} STATIC ${ARGN})
  target_sources(${target} PUBLIC FILE_SET userfs_modules TYPE CXX_MODULES
    BASE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}" FILES "${CMAKE_CURRENT_SOURCE_DIR}/${interface}")
  set_property(TARGET ${target} PROPERTY CXX_SCAN_FOR_MODULES ON)
  target_link_libraries(${target} PUBLIC xv_options)
endfunction()
