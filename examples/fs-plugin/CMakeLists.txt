cmake_minimum_required(VERSION 3.28)
project(hello_fs_plugin LANGUAGES C ASM)
get_filename_component(default_sdk "${CMAKE_CURRENT_SOURCE_DIR}/../../user/fs/sdk" ABSOLUTE)
set(XVFS_SDK "${default_sdk}" CACHE PATH "Standalone XVFS SDK directory")
set(XVFS_PLUGIN_SOURCE "hello.c" CACHE FILEPATH "Plugin source file")
add_executable(hello "${XVFS_PLUGIN_SOURCE}" "${XVFS_SDK}/standalone/crt0.S" "${XVFS_SDK}/standalone/memory.c")
target_include_directories(hello PRIVATE "${XVFS_SDK}/include")
target_compile_features(hello PRIVATE c_std_11)
target_compile_options(hello PRIVATE -O2 -Wall -Wextra -Werror)
target_link_options(hello PRIVATE -z max-page-size=4096 -T "${XVFS_SDK}/standalone/plugin.ld")
set_target_properties(hello PROPERTIES SUFFIX .elf LINK_DEPENDS "${XVFS_SDK}/standalone/plugin.ld")
configure_file(hello.conf hello.conf COPYONLY)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/install.tsv" CONTENT
  "dir\t0755\t/usr\ndir\t0755\t/usr/lib\ndir\t0755\t/usr/lib/fsplugins\ndir\t0755\t/etc\ndir\t0755\t/etc/fsplugins\nfile\t0755\t/usr/lib/fsplugins/hello.elf\t$<TARGET_FILE:hello>\nfile\t0644\t/etc/fsplugins/hello.conf\t${CMAKE_CURRENT_BINARY_DIR}/hello.conf\n")
